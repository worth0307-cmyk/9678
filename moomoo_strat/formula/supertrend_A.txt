{SuperTrend 精确版：ATR 10、倍数 3、Source=hl2，和 TradingView 逐根一致。需要 moomoo 允许变量引用自己上一根的值；报错就用 supertrend_B.txt}
TR1:=MAX(MAX(HIGH-LOW,ABS(HIGH-REF(CLOSE,1))),ABS(LOW-REF(CLOSE,1)));
ATRV:=EMA(TR1,19);
UP0:=(HIGH+LOW)/2-3*ATRV;
DN0:=(HIGH+LOW)/2+3*ATRV;
UPL:=IF(REF(CLOSE,1)>REF(UPL,1),MAX(UP0,REF(UPL,1)),UP0);
DNL:=IF(REF(CLOSE,1)<REF(DNL,1),MIN(DN0,REF(DNL,1)),DN0);
TRD:=IF(REF(TRD,1)=-1,IF(CLOSE>REF(DNL,1),1,-1),IF(CLOSE<REF(UPL,1),-1,1));
ST:IF(TRD=1,UPL,DNL),COLORYELLOW,LINETHICK2;
DRAWTEXT(TRD=1 AND REF(TRD,1)=-1,LOW*0.97,'买'),COLORGREEN;
DRAWTEXT(TRD=-1 AND REF(TRD,1)=1,HIGH*1.03,'卖'),COLORRED;
