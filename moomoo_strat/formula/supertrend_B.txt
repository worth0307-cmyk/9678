{SuperTrend 近似版：不用自引用，只用 HHV/LLV/BARSLAST。和 TradingView 约 97% 的K线方向一致，翻转点偶尔早一两根或晚一两根}
TR1:=MAX(MAX(HIGH-LOW,ABS(HIGH-REF(CLOSE,1))),ABS(LOW-REF(CLOSE,1)));
ATRV:=EMA(TR1,19);
U1:=HHV((HIGH+LOW)/2-3*ATRV,20);
D1:=LLV((HIGH+LOW)/2+3*ATRV,20);
TRD:=IF(BARSLAST(CLOSE>REF(D1,1))<BARSLAST(CLOSE<REF(U1,1)),1,-1);
ST:IF(TRD=1,U1,D1),COLORYELLOW,LINETHICK2;
DRAWTEXT(TRD=1 AND REF(TRD,1)=-1,LOW*0.97,'买'),COLORGREEN;
DRAWTEXT(TRD=-1 AND REF(TRD,1)=1,HIGH*1.03,'卖'),COLORRED;
